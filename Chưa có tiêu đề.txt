local KEY_TAIL     = ".  "
local KEY_DURATION = 3600
local KEY_FILE     = "HM_key65.dat"
local unlocked = false

local function readKeyRaw()
    local ok, d = pcall(function()
        if readfile and isfile and isfile(KEY_FILE) then return readfile(KEY_FILE) end
        return nil
    end)
    return ok and d or nil
end
local function writeKeyRaw(s) 
    pcall(function() if writefile then writefile(KEY_FILE, s) end end) 
end
local function wipeKey() 
    pcall(function() if writefile then writefile(KEY_FILE, "") end end) 
end
local function validateKey(k)
    if type(k) ~= "string" or #k < 4 then return false end
    if string.sub(k, -3) ~= KEY_TAIL then return false end
    return true
end
local function keyValidNow()
    local raw = readKeyRaw()
    if not raw or raw == "" then return false end
    local a, b = string.find(raw, "|", 1, true)
    if not a then return false end
    local k  = string.sub(raw, 1, a - 1)
    local ts = tonumber(string.sub(raw, b + 1))
    if not k or not ts or not validateKey(k) then return false end
    if (os.time() - ts) >= KEY_DURATION then return false end
    return true
end

local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then return false end
    local c = p.Character
    if not c then return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then return nil end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), name, 1, true) then return p end
    end
    return nil
end

local function IsMonster(model)
    if not model or not model:IsA("Model") then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if model == LocalPlayer.Character then return false end
    local plr = Players:GetPlayerFromCharacter(model)
    if plr then return false end
    local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
    if not root then return false end
    return true
end

local function GetMonsterRoot(model)
    return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
end

local function ScanMonsters()
    local list = {}
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Model") and IsMonster(v) then
            table.insert(list, v)
        end
    end
    return list
end

local function FindTargetByNamePartial(partial)
    if not partial or partial == "" then return nil end
    local low = string.lower(partial)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then
            return p
        end
    end
    return nil
end

local function CollectAllMonsters()
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        if root then
            local angle = (count / math.max(1, #found)) * math.pi * 2
            local pos = myRoot.Position + Vector3.new(
                math.cos(angle) * Config.MonsterOrbitRadius,
                Config.MonsterOrbitHeight,
                math.sin(angle) * Config.MonsterOrbitRadius
            )
            pcall(function()
                model:PivotTo(CFrame.new(pos))
                local hum = model:FindFirstChildOfClass("Humanoid")
                if hum then hum.WalkSpeed = 0 end
            end)
            monsters[model] = true
            count = count + 1
        end
    end
end

local function SummonAgainstTarget()
    local target = Config.MonsterTarget
    if not target or not target.Character then return end
    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
    if not tRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        local hum = model:FindFirstChildOfClass("Humanoid")
        if root and hum then
            if Config.MonsterMode == "TP" then
                local offset = Vector3.new(math.random(-4, 4), 2, math.random(-4, 4))
                pcall(function() model:PivotTo(CFrame.new(tRoot.Position + offset)) end)
            end
            monsters[model] = true
            count = count + 1
        end
    end
end

-- ==========================================================
task.spawn(function()
    while true do
        task.wait(0.02)
        if State.ORLock and not State.Farm then
            pcall(function()
                if not Valid(Config.Target) then
                    local name = Config.TargetName
                    if not name or name == "" then
                        name = TargetBox.Text
                        Config.TargetName = name
                    end
                    if name == "" then Config.Target = FindNearest(Config.ORRange)
                    else Config.Target = FindByName(name) end
                    if not Config.Target then ORLastTarget = nil return end
                end
                if ORLastTarget ~= Config.Target then ORLastTarget = Config.Target ResetDash() end
                local tc = Config.Target.Character
                local tRoot = tc and tc:FindFirstChild("HumanoidRootPart")
                local myC = LocalPlayer.Character
                local myRoot = myC and myC:FindFirstChild("HumanoidRootPart")
                local myHum = myC and myC:FindFirstChildOfClass("Humanoid")
                if not (tRoot and myRoot and myHum) then return end
                local dist = (myRoot.Position - tRoot.Position).Magnitude
                if dist > 2000 then return end
                local tp = tRoot.Position

                if Config.ORMode == "DASH" then
                    local now = tick()
                    local function groundAt(x, z, baseY)
                        local rp = RaycastParams.new()
                        rp.FilterType = Enum.RaycastFilterType.Exclude
                        rp.FilterDescendantsInstances = {myC, tc}
                        local hit = workspace:Raycast(Vector3.new(x, baseY + 20, z), Vector3.new(0, -80, 0), rp)
                        return hit and hit.Position.Y or (baseY - 3)
                    end
                    local function tpToTarget()
                        local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(Vector3.new(behind.X, tp.Y, behind.Z), tp)
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                    end
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        dashOrbitAngle = math.random() * math.pi * 2
                        tpToTarget()
                        dashPhase = "attack"
                        dashTimer = now + Config.DashAttackDuration
                        dashNextHit = 0
                    elseif dashPhase == "attack" then
                        dashOrbitAngle = dashOrbitAngle + math.rad(Config.DashOrbitSpeed) * 0.02
                        local px = tp.X + math.cos(dashOrbitAngle) * Config.DashOrbitRadius
                        local pz = tp.Z + math.sin(dashOrbitAngle) * Config.DashOrbitRadius
                        local gy = groundAt(px, pz, tp.Y)
                        local desired = Vector3.new(px, gy + Config.DashGroundOffset, pz)
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(desired, Vector3.new(tp.X, desired.Y, tp.Z))
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                        if now >= dashNextHit then
                            dashNextHit = now + Config.DashHitRate
                            HitAll(Config.Target)
                        end
                        if now >= dashTimer then
                            if dashSavedPos then
                                pcall(function()
                                    myRoot.CFrame = dashSavedPos
                                    myRoot.AssemblyLinearVelocity = Vector3.zero
                                end)
                            end
                            dashPhase = "wait"
                            dashTimer = now + Config.DashWaitDuration
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then
                            dashOrbitAngle = math.random() * math.pi * 2
                            tpToTarget()
                            dashPhase = "attack"
                            dashTimer = now + Config.DashAttackDuration
                            dashNextHit = 0
                        end
                    end
                elseif Config.ORMode == "EZY" then
                    local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                    local dir = behind - myRoot.Position
                    if dir.Magnitude > 1.5 then
                        myRoot.AssemblyLinearVelocity = dir.Unit * (Config.ORWalkSpeed * 10)
                    else
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        pcall(function() myRoot.CFrame = CFrame.lookAt(myRoot.Position, tp) end)
                    end
                    EzyFastAttack(Config.Target)
                else
                    local fp
                    local dcf = tRoot.CFrame
                    if Config.ORMode == "FOLLOW" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "BEHIND" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "ORBIT" then
                        Config.orbitAngle = Config.orbitAngle + math.rad(Config.ORRotSpeed) * 0.05
                        fp = tp + Vector3.new(math.cos(Config.orbitAngle)*Config.ORDistance, Config.ORHeight, math.sin(Config.orbitAngle)*Config.ORDistance)
                    else fp = tp - dcf.LookVector * Config.ORDistance end
                    pcall(function()
                        myRoot.CFrame = CFrame.lookAt(fp + Vector3.new(0, Config.ORHeight, 0), tp + Vector3.new(0, Config.ORHeight, 0))
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                    end)
                    if dist <= Config.ORNear then HitAll(Config.Target) end
                end
            end)
        end
    end
end)

-- MONSTER LOOP
task.spawn(function()
    while true do
        task.wait(0.05)

            if State.MonsterCollect and next(monsters) == nil then
                CollectAllMonsters()
            end
            local myChar = LocalPlayer.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then return end
            local target = Config.MonsterTarget
            local tRoot = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            monsterOrbitAngle = monsterOrbitAngle + math.rad(Config.MonsterOrbitSpeed) * 0.05
            local i = 0
            local totalMon = 0
            for m in pairs(monsters) do totalMon = totalMon + 1 end
            for model, _ in pairs(monsters) do
                i = i + 1
                if not model.Parent or not IsMonster(model) then
                    monsters[model] = nil
                    monsterAttackTime[model] = nil
                else
                    local root = GetMonsterRoot(model)
                    local hum = model:FindFirstChildOfClass("Humanoid")
                    if root and hum then
                        if State.MonsterSummon and Config.MonsterMode ~= "OFF" and tRoot then
                            local dist = (root.Position - tRoot.Position).Magnitude
                            if Config.MonsterMode == "TP" then
                                if dist > 10 then
                                    local off = Vector3.new(math.random(-4, 4), 2, math.random(-4, 4))
                                    pcall(function() model:PivotTo(CFrame.new(tRoot.Position + off)) end)
                                end
                            elseif Config.MonsterMode == "WALK" then
                                if dist > 3 then
                                    pcall(function()
                                        hum.WalkSpeed = 25 * Config.MonsterFollowSpeed
                                        hum:MoveTo(tRoot.Position)
                                    end)
                                else
                                    pcall(function() hum.WalkSpeed = 0 end)
                                end
                            end
                            if State.MonsterAttack and dist <= 8 then
                                local now = tick()
                                local last = monsterAttackTime[model] or 0
                                if now - last >= Config.MonsterAttackCD then
                                    monsterAttackTime[model] = now
                                    local th = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
                                    if th and th.Health > 0 then
                                        pcall(function() th:TakeDamage(15) end)
                                        pcall(function()
                                            local cr = ReplicatedStorage:FindFirstChild("CombatRemote")
                                            if cr then cr:FireServer("M1", target) end
                                        end)
                                        for _, r in ipairs(ReplicatedStorage:GetDescendants()) do
                                            if (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then
                                                local n = string.lower(r.Name)
                                                if n:find("attack") or n:find("damage") or n:find("hit") then
                                                    pcall(function() r:FireServer(target, 15) end)
                                                    pcall(function() r:FireServer(target) end)
                                                end
                                            end
                                        end
                                    end
                                end
                            end
                        end
                        if State.MonsterOrbit and (not State.MonsterSummon or Config.MonsterMode == "OFF" or not tRoot) then
                            local angle = monsterOrbitAngle + (i / math.max(1, totalMon)) * math.pi * 2
                            local pos = myRoot.Position + Vector3.new(
                                math.cos(angle) * Config.MonsterOrbitRadius,
                                Config.MonsterOrbitHeight,
                                math.sin(angle) * Config.MonsterOrbitRadius
                            )

                                model:PivotTo(CFrame.new(pos, myRoot.Position))
                                hum.WalkSpeed = 0
                            end)
                        end
                    end
                end
            end
        end)
    end
end)

                            end
                        end
                        if m1Detected then
                            lastM1Time = tick()
                            rawBlock()
                        else
                            if isBlocking and (tick() - lastM1Time) > BUFFER_TIME then rawUnblock() end
                        end
                    end
                end
            end
        end
        if Config.AutoReset then
            local myChar = LocalPlayer.Character
            local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
            if myHum and myHum.Health > 0 then
                if State.ORLock or State.M1Spam or State.Spam then
                    if tick() - lastResetTimer >= Config.ResetTime then
                        if resetEvent then pcall(function() resetEvent:FireServer() end) end
                        lastResetTimer = tick()
                    end
                else lastResetTimer = tick() end
            else lastResetTimer = tick() end
        end
        if State.Nametag then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local head = p.Character:FindFirstChild("Head")
                    local hum = p.Character:FindFirstChildOfClass("Humanoid")
                    if head and hum then
                        local tag = head:FindFirstChild("NameTagGUI")
                        if not tag then
                            tag = Instance.new("BillboardGui")
                            tag.Name = "NameTagGUI"
                            tag.Adornee = head
                            tag.Size = UDim2.new(0, 200, 0, 50)
                            tag.StudsOffset = Vector3.new(0, 3, 0)
                            tag.AlwaysOnTop = true
                            tag.Parent = head
                            local lbl = Instance.new("TextLabel", tag)
                            lbl.Size = UDim2.new(1, 0, 1, 0)
                            lbl.BackgroundTransparency = 1
                            lbl.TextScaled = true
                            lbl.TextColor3 = Color3.fromRGB(0, 255, 100)
                            lbl.Font = Enum.Font.GothamBold
                        end
                        local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                        local tr = p.Character:FindFirstChild("HumanoidRootPart")
                        local d = mr and tr and (mr.Position - tr.Position).Magnitude or 0
                        local lbl = tag:FindFirstChildOfClass("TextLabel")
                        if lbl then
                            lbl.Text = p.Name .. " | " .. math.floor(hum.Health) .. "HP | " .. math.floor(d) .. "m"
                        end
                    end
                end
            end
        else
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character then
                    local head = p.Character:FindFirstChild("Head")
                    if head then
                        local tag = head:FindFirstChild("NameTagGUI")
                        if tag then tag:Destroy() end
                    end
                end
            end
        end
        if State.Spin then
            local c = LocalPlayer.Character
            if c then
                local hrp = c:FindFirstChild("HumanoidRootPart")
                if hrp then hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(15), 0) end
            end
        end
        if State.AntiStun then
            local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if h then
                if h.PlatformStand then h.PlatformStand = false end
                if h.Sit then h.Sit = false end
                h.AutoRotate = true
            end
        end
        if State.Dash then
            local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if h then
                local target = Config.AntiDetect and 60 or 200
                if h.WalkSpeed ~= target then h.WalkSpeed = target end
                if h.JumpPower ~= 80 then h.JumpPower = 80 end
            end
        end
        if State.GodMode then
            local c = LocalPlayer.Character
            if c then
                local h = c:FindFirstChildOfClass("Humanoid")
                if h then h.MaxHealth = math.huge; h.Health = math.huge end
                if State.NoHunger then
                    for _, s in ipairs(c:GetDescendants()) do
                        if s:IsA("NumberValue") then
                            local n = s.Name:lower()
                            if n:find("hunger") or n:find("thirst") or n:find("cold") or n:find("temperature") then
                                s.Value = 100
                            end
                        end
                    end
                end
            end
        end
        if State.NoClip then
            local c = LocalPlayer.Character
            if c then
                for _, part in ipairs(c:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
                local hrp = c:FindFirstChild("HumanoidRootPart")
                if hrp and hrp.Position.Y < -50 then
                    hrp.CFrame = CFrame.new(hrp.Position.X, 50, hrp.Position.Z)
                    hrp.AssemblyLinearVelocity = Vector3.zero
                end
            end
        end
        if State.Aim and Config.ORMode ~= "DASH" and Config.ORMode ~= "EZY" then
            local t = State.ORLock and Valid(Config.Target) and Config.Target or FindNearest(Config.AttackRange)
            if t then
                local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
                if mr and tr then mr.CFrame = CFrame.lookAt(mr.Position, tr.Position) end
            end
        end
        if State.TP and not State.ORLock then
            local t = FindNearest(Config.AttackRange)
            if t then
                local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
                if mr and tr then 
                    mr.CFrame = tr.CFrame * CFrame.new(0, 0, Config.TPDistance)
                    tpTarget = t
                    tpStartTime = tick()
                end
            end
        end
        if State.TP and tpTarget then
            local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            local tr = tpTarget.Character and tpTarget.Character:FindFirstChild("HumanoidRootPart")
            if mr and tr then
                if tick() - tpStartTime < 0.5 then
                    mr.CFrame = tr.CFrame * CFrame.new(0, 0, Config.TPDistance)
                else
                    State.TP = false
                    tpTarget = nil
                end
            else
                State.TP = false
                tpTarget = nil
            end
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if Valid(p) then
                local pc = p.Character
                if State.ESP then
                    if not pc:FindFirstChild("ESP_HL") then
                        local hl = Instance.new("Highlight")
                        hl.Name = "ESP_HL"
                        hl.FillColor = Color3.fromRGB(0, 255, 100)
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.5
                        hl.Parent = pc
                    end
                else
                    local hl = pc:FindFirstChild("ESP_HL")
                    if hl then hl:Destroy() end
                end
            end
        end
    end)
end)
